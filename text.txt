<script>
  // ====== tiny demo chart (no libs) ======
  const canvas = document.getElementById('chartCanvas');
  const ctx = canvas.getContext('2d');

  const ranges = { "24H": 40, "7D": 60, "30D": 90, "90D": 120, "1Y": 160 };

  function makeSeries(n){
    const base = 0.095;
    let v = base;
    const data = [];
    for(let i=0;i<n;i++){
      const drift = Math.sin(i/6)*0.0006;
      const noise = (Math.random()-0.5)*0.0042;
      v = Math.max(0.02, v + drift + noise);
      data.push(v);
    }
    return data;
  }

  function drawChart(data){
    const dpr = window.devicePixelRatio || 1;
    const cssW = canvas.clientWidth;
    const cssH = canvas.clientHeight;
    canvas.width = Math.floor(cssW * dpr);
    canvas.height = Math.floor(cssH * dpr);
    ctx.setTransform(dpr,0,0,dpr,0,0);

    const w = cssW, h = cssH, pad = 16;
    ctx.clearRect(0,0,w,h);

    let min = Math.min(...data);
    let max = Math.max(...data);
    const extra = (max-min) * 0.15 || 0.01;
    min -= extra; max += extra;

    document.getElementById('hiText').textContent = '$' + (Math.max(...data)).toFixed(3);
    document.getElementById('loText').textContent = '$' + (Math.min(...data)).toFixed(3);

    // grid
    ctx.lineWidth = 1;
    ctx.strokeStyle = 'rgba(255,255,255,0.08)';
    for(let r=1;r<5;r++){
      const y = pad + (h-2*pad) * (r/5);
      ctx.beginPath(); ctx.moveTo(pad,y); ctx.lineTo(w-pad,y); ctx.stroke();
    }
    for(let c=1;c<7;c++){
      const x = pad + (w-2*pad) * (c/7);
      ctx.beginPath(); ctx.moveTo(x,pad); ctx.lineTo(x,h-pad); ctx.stroke();
    }

    // neon line
    ctx.lineWidth = 3;
    ctx.strokeStyle = 'rgba(124,255,107,0.95)';
    ctx.shadowBlur = 12;
    ctx.shadowColor = 'rgba(124,255,107,0.35)';
    ctx.beginPath();
    data.forEach((v,i)=>{
      const x = pad + (w-2*pad) * (i/(data.length-1));
      const y = pad + (h-2*pad) * (1 - ((v-min)/(max-min)));
      if(i===0) ctx.moveTo(x,y); else ctx.lineTo(x,y);
    });
    ctx.stroke();

    // fill
    ctx.shadowBlur = 0;
    ctx.lineTo(w-pad, h-pad);
    ctx.lineTo(pad, h-pad);
    ctx.closePath();
    const grad = ctx.createLinearGradient(0,pad,0,h-pad);
    grad.addColorStop(0,'rgba(124,255,107,0.14)');
    grad.addColorStop(1,'rgba(255,79,216,0.03)');
    ctx.fillStyle = grad;
    ctx.fill();
  }

  let current = "24H";
  let series = makeSeries(ranges[current]);
  drawChart(series);

  document.querySelectorAll('.tab').forEach(tab=>{
    tab.addEventListener('click', ()=>{
      document.querySelectorAll('.tab').forEach(t=>{
        t.classList.remove('active');
        t.setAttribute('aria-selected','false');
      });
      tab.classList.add('active');
      tab.setAttribute('aria-selected','true');

      current = tab.dataset.range;
      series = makeSeries(ranges[current]);
      drawChart(series);
    });
  });

  window.addEventListener('resize', ()=> drawChart(series));
  document.getElementById('year').textContent = new Date().getFullYear();
</script>
